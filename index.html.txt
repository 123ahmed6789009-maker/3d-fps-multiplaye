        <h1>🚀 لعبة القتال الجماعية 3D</h1>
        <p>انقر للبدء في القتال واللعب أونلاين عبر المتصفح</p>

    <script>
        const socket = io();
        const otherPlayers = {};

        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x111122);
        const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        document.getElementById('canvas-container').appendChild(renderer.domElement);

        const ambientLight = new THREE.AmbientLight(0xffffff, 0.7);
        scene.add(ambientLight);

        const floor = new THREE.Mesh(
            new THREE.PlaneGeometry(80, 80),
            new THREE.MeshStandardMaterial({ color: 0x2d3436 })
        );
        floor.rotation.x = -Math.PI / 2;
        scene.add(floor);

        camera.position.set(0, 1.6, 0);

        function startGame() {
            document.getElementById('intro-screen').style.display = 'none';
            document.body.requestPointerLock();
        }

        socket.on('currentPlayers', (players) => {
            Object.keys(players).forEach((id) => {
                if (id !== socket.id) addOtherPlayer(id, players[id]);
            });
        });

        socket.on('newPlayer', (data) => addOtherPlayer(data.id, data.player));

        socket.on('playerMoved', (data) => {
            if (otherPlayers[data.id]) {
                otherPlayers[data.id].position.set(data.player.x, data.player.y, data.player.z);
            }
        });

        socket.on('playerDisconnected', (id) => {
            if (otherPlayers[id]) {
                scene.remove(otherPlayers[id]);
                delete otherPlayers[id];
            }
        });

        function addOtherPlayer(id, data) {
            const mesh = new THREE.Mesh(
                new THREE.CylinderGeometry(0.5, 0.5, 1.6, 16),
                new THREE.MeshStandardMaterial({ color: data.color })
            );
            mesh.position.set(data.x, data.y, data.z);
            scene.add(mesh);
            otherPlayers[id] = mesh;
        }

        const keys = {};
        window.addEventListener('keydown', e => keys[e.key.toLowerCase()] = true);
        window.addEventListener('keyup', e => keys[e.key.toLowerCase()] = false);

        function animate() {
            requestAnimationFrame(animate);

            let moved = false;
            if (keys['w']) { camera.position.z -= 0.1; moved = true; }
            if (keys['s']) { camera.position.z += 0.1; moved = true; }
            if (keys['a']) { camera.position.x -= 0.1; moved = true; }
            if (keys['d']) { camera.position.x += 0.1; moved = true; }

            if (moved) {
                socket.emit('playerMove', { x: camera.position.x, y: camera.position.y, z: camera.position.z });
            }

            renderer.render(scene, camera);
        }
        animate();
    </script>
